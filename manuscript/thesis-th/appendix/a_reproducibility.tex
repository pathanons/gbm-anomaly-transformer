\chapter{Reproducibility Notes}

\section{คำสั่งการรันบน macOS/Linux}

\begin{verbatim}
DEVICE=auto bash run.sh --visualize
\end{verbatim}

\section{คำสั่งการรันบน Windows}

\begin{verbatim}
run.bat --visualize
\end{verbatim}

\section{Mac Apple Silicon}

\begin{verbatim}
conda env create -f environment.macos-mps.yml
conda activate gbm-anomaly-transformer-mps
DEVICE=mps bash run.sh --visualize
\end{verbatim}
